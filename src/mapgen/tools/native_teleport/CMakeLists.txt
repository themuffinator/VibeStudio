cmake_minimum_required(VERSION 3.28)
project(qmg_native_teleport_reference LANGUAGES C)
set(Q2_REFERENCE_SOURCE "E:/_SOURCE/_CODE/Quake-2-master" CACHE PATH "Read-only official Quake II 3.19 source")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/reference.json" reference_json)
string(JSON count LENGTH "${reference_json}" dependencies)
math(EXPR last "${count}-1")
set(external_c)
foreach(index RANGE 0 ${last})
    string(JSON name GET "${reference_json}" dependencies ${index} relative_path)
    string(JSON expected GET "${reference_json}" dependencies ${index} sha256)
    file(SHA256 "${Q2_REFERENCE_SOURCE}/${name}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned external file changed: ${name}")
    endif()
    if(name MATCHES "\\.c$")
        list(APPEND external_c "${Q2_REFERENCE_SOURCE}/${name}")
    endif()
endforeach()
# All game, movement and server-area/trace code is an unmodified external input.
# This optional private combined executable is GPL; it is never a map artifact.
add_executable(qmg_native_teleport host.c server_bridge.c ${external_c})
target_include_directories(qmg_native_teleport PRIVATE "${Q2_REFERENCE_SOURCE}/game" "${Q2_REFERENCE_SOURCE}/qcommon" "${Q2_REFERENCE_SOURCE}/server")
target_compile_definitions(qmg_native_teleport PRIVATE C_ONLY GAME_HARD_LINKED _CRT_SECURE_NO_WARNINGS)
target_compile_options(qmg_native_teleport PRIVATE -Wno-error=implicit-int)
if(MSVC)
    target_compile_options(qmg_native_teleport PRIVATE /fp:strict /Od)
else()
    target_compile_options(qmg_native_teleport PRIVATE -O0 -ffp-contract=off)
    target_link_libraries(qmg_native_teleport PRIVATE m)
endif()
# Deliberately no install or package target.
